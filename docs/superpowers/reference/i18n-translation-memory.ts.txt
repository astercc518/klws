export type MessageNS = keyof typeof messages;

/** Per-namespace key union — reuse this instead of re-declaring a local
 *  `keyof (typeof messages)[N]` alias in each page/helper. */
export type NSKey<N extends MessageNS> = keyof (typeof messages)[N];

/** Curried getter: `t(locale)("nav","language")`. Key is type-checked
 *  against the namespace, so typos fail at compile time. */
export function t(locale: Locale) {
  return <N extends MessageNS>(ns: N, key: NSKey<N>): string =>
    (messages[ns][key] as Entry)[locale];
}
